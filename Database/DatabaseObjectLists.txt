-- Drop all other tables except ones used in Sp_GetUserModuleTabHierarchy & Sp_GetMenuForTab
DECLARE @sql NVARCHAR(MAX) = N'';

SELECT @sql += 'DROP TABLE IF EXISTS [' + s.name + '].[' + t.name + '];' + CHAR(13)
FROM sys.tables t
JOIN sys.schemas s ON t.schema_id = s.schema_id
WHERE t.name NOT IN (
    '_AppPage',
    '_AppPageTab',
    '_AppPageTabPermission',
    '_AppPageGrid',
    '_AppPageGridPermission',
    '_AppFormTabGrid',
    '_AppFormTabGridPermission',
    '_AppRolePermission',
    '_AppRole',
    '_AppUserRole',
    '_AppUser'
)
ORDER BY t.name;

-- 2) Tables to KEEP (used by Sp_GetUserModuleTabHierarchy / Sp_GetMenuForTab)
DECLARE @Keep TABLE (Tbl sysname PRIMARY KEY);
INSERT INTO @Keep (Tbl) VALUES
(N'AppTabsMaster'),
(N'UserModuleTabsAccess'),
(N'tblButtonAction');

-- 3) Compute existing drop targets from DB objects (exclude keepers)
DECLARE @ToDrop TABLE (Tbl sysname PRIMARY KEY);
INSERT INTO @ToDrop (Tbl)
SELECT t.name
FROM sys.tables t
WHERE SCHEMA_NAME(t.schema_id) = 'dbo'
  AND t.name IN (SELECT Tbl FROM @AllScriptTables)
  AND t.name NOT IN (SELECT Tbl FROM @Keep);

-- 4) Build dynamic DROP statements with IF EXISTS guard
DECLARE @sql nvarchar(max) = N'';

SELECT @sql = STRING_AGG(
    N'IF OBJECT_ID(N''[dbo].[' + Tbl + N']'', N''U'') IS NOT NULL DROP TABLE [dbo].[' + Tbl + N'];',
    CHAR(10)
)
FROM @ToDrop;


-- 4) Drop each table individually (no STRING_AGG, no FK handling)
DECLARE @tbl sysname, @sql nvarchar(max);

DECLARE cur CURSOR LOCAL FAST_FORWARD FOR
  SELECT Tbl FROM @ToDrop ORDER BY Tbl;

OPEN cur;
FETCH NEXT FROM cur INTO @tbl;
WHILE @@FETCH_STATUS = 0
BEGIN
  SET @sql = N'IF OBJECT_ID(N''[dbo].[' + @tbl + N']'', N''U'') IS NOT NULL
               DROP TABLE [dbo].[' + @tbl + N'];';
  PRINT @sql;  -- optional: log what’s being executed
  EXEC sys.sp_executesql @sql;

  FETCH NEXT FROM cur INTO @tbl;
END
CLOSE cur; DEALLOCATE cur;


-- 5) Review or run
PRINT N'-- The following DROP statements will be executed:';
PRINT @sql;

EXEC sys.sp_executesql @sql;

-- (Optional) Show what remained vs considered
PRINT N'-- Kept tables (not dropped):';
SELECT Tbl FROM @Keep;
PRINT N'-- Dropped tables (existed and were in the script, excluding keep):';
SELECT Tbl FROM @ToDrop;
