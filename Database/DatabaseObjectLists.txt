-- Drop all other tables except ones used in Sp_GetUserModuleTabHierarchy & Sp_GetMenuForTab
DECLARE @sql NVARCHAR(MAX) = N'';

SELECT @sql += 'DROP TABLE IF EXISTS [' + s.name + '].[' + t.name + '];' + CHAR(13)
FROM sys.tables t
JOIN sys.schemas s ON t.schema_id = s.schema_id
WHERE t.name NOT IN (
    '_AppPage',
    '_AppPageTab',
    '_AppPageTabPermission',
    '_AppPageGrid',
    '_AppPageGridPermission',
    '_AppFormTabGrid',
    '_AppFormTabGridPermission',
    '_AppRolePermission',
    '_AppRole',
    '_AppUserRole',
    '_AppUser'
)
ORDER BY t.name;
